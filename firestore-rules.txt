rules_version = '2';
// ═══════════════════════════════════════════════════════════════
// RespirApp — Regras Firestore ENDURECIDAS (v43)
// Princípio: negar por padrão; cada usuário só enxerga e grava o que é dele;
// o admin só faz o estritamente necessário; ninguém altera o próprio plano.
// ═══════════════════════════════════════════════════════════════
service cloud.firestore {
  match /databases/{database}/documents {

    // ── Funções auxiliares ──────────────────────────────────────
    function signedIn() { return request.auth != null; }

    // Admin identificado pelo e-mail do token (mesmo e-mail usado no app).
    // OPCIONAL, mais forte: troque por  request.auth.uid == 'UID_DO_ADMIN'
    // (Console Firebase > Authentication > copie o "UID de usuário" do admin).
    function isAdmin() {
     return signedIn() && request.auth.uid == 'fs5Prj7mqyaMR0BYRVJFT7zjqWr1';
   }

    function isOwner(uid) { return signedIn() && request.auth.uid == uid; }

    function strMax(v, n) { return v is string && v.size() <= n; }

    function userLevel() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.get('accessLevel', 'gratuito');
    }

    // ── users ───────────────────────────────────────────────────
    match /users/{userId} {
      // Cadastro: o usuário NÃO pode se criar já com plano pago (sem accessLevel).
      allow create: if isOwner(userId)
        && request.resource.data.keys().hasOnly(['name','email','whatsapp','createdAt','consentAccepted','questionnaireCompleted'])
        && request.resource.data.keys().hasAll(['name','email','whatsapp','createdAt','consentAccepted','questionnaireCompleted'])
        && request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 80
        && strMax(request.resource.data.whatsapp, 25)
        && request.resource.data.email is string
        && request.resource.data.email.lower() == request.auth.token.email.lower()
        && request.resource.data.consentAccepted == false
        && request.resource.data.questionnaireCompleted == false
        && request.resource.data.createdAt == request.time;

      allow read: if isOwner(userId) || isAdmin();

      // Usuário: só pode marcar consentimento e conclusão do questionário.
      // NUNCA pode alterar accessLevel, email, name etc.
      allow update: if
        (isOwner(userId)
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['consentAccepted','consentAt','questionnaireCompleted'])
          && request.resource.data.consentAccepted == true)
        ||
        // Admin: só pode mudar o plano, e apenas para valores válidos.
        (isAdmin()
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['accessLevel'])
          && request.resource.data.accessLevel in ['gratuito','grupo','mentoria']);

      allow delete: if isAdmin();
    }

    // ── questionnaires ──────────────────────────────────────────
    match /questionnaires/{userId} {
      allow create, update: if isOwner(userId)
        && request.resource.data.keys().hasOnly(['q1','q2','q3','q4','q5','q7','submittedAt']);
      allow read: if isOwner(userId) || isAdmin();
    }

    // ── prontuarios (somente plano Mentoria) ────────────────────
    match /prontuarios/{userId} {
      allow read: if isOwner(userId) || isAdmin();

      allow create: if isOwner(userId)
        && (userLevel() == 'mentoria' || isAdmin())
        && request.resource.data.keys().hasOnly(['userId','data','locked','updatedAt','createdAt'])
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.data is map
        && request.resource.data.locked == false;

      allow update: if
        // Usuário edita só enquanto NÃO estiver bloqueado e não consegue se desbloquear
        (isOwner(userId)
          && (userLevel() == 'mentoria' || isAdmin())
          && resource.data.get('locked', false) == false
          && request.resource.data.keys().hasOnly(['userId','data','locked','updatedAt','createdAt'])
          && request.resource.data.userId == request.auth.uid
          && request.resource.data.data is map
          && request.resource.data.locked == false)
        ||
        // Admin só alterna o bloqueio
        (isAdmin()
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['locked']));
    }

    // ── sessions ────────────────────────────────────────────────
    // Antes: qualquer usuário logado lia as sessões de TODOS. Agora: só as próprias (admin lê tudo).
    match /sessions/{sessionId} {
      allow create: if signedIn()
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.type in ['breathe','audio','test']
        && request.resource.data.keys().hasOnly([
             'userId','type','protocol','protocolName','totalCycles','startedAt','endedAt','completedAt',
             'wellbeingBefore','wellbeingAfter','level','audioTitle','testName','testLabel','times','average']);
      allow read: if signedIn() && (resource.data.userId == request.auth.uid || isAdmin());
      allow update, delete: if false;
    }

    // ── habits ──────────────────────────────────────────────────
    match /habits/{habitId} {
      allow create: if signedIn()
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.name is string
        && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 60
        && strMax(request.resource.data.get('question', ''), 200)
        && strMax(request.resource.data.get('notes', ''), 1000)
        && strMax(request.resource.data.get('unit', ''), 20);
      allow read: if signedIn() && (resource.data.userId == request.auth.uid || isAdmin());
      allow update: if signedIn()
        && resource.data.userId == request.auth.uid
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.name is string
        && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 60
        && strMax(request.resource.data.get('question', ''), 200)
        && strMax(request.resource.data.get('notes', ''), 1000)
        && strMax(request.resource.data.get('unit', ''), 20);
      allow delete: if signedIn() && resource.data.userId == request.auth.uid;
    }

    // ── habitLogs ───────────────────────────────────────────────
    match /habitLogs/{logId} {
      allow create: if signedIn()
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.keys().hasOnly(['userId','habitId','date','completed','value','timestamp']);
      allow read: if signedIn() && (resource.data.userId == request.auth.uid || isAdmin());
      allow update: if signedIn()
        && resource.data.userId == request.auth.uid
        && request.resource.data.userId == request.auth.uid;
      allow delete: if signedIn() && resource.data.userId == request.auth.uid;
    }

    // ── Tudo o que não foi listado acima: NEGADO ────────────────
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
